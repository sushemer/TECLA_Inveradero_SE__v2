cmake_minimum_required(VERSION 3.16)
set(TECLA_NODO_ID "2" CACHE STRING "Perfil físico del nodo (2 o 4)")
if(NOT TECLA_NODO_ID STREQUAL "2" AND NOT TECLA_NODO_ID STREQUAL "4")
    message(FATAL_ERROR "TECLA_NODO_ID debe ser 2 o 4 (recibido '${TECLA_NODO_ID}')")
endif()
if(TECLA_NODO_ID STREQUAL "2")
    set(_TECLA_DEFAULT_SENSOR_PROFILE "DHT22_VEML7700")
else()
    set(_TECLA_DEFAULT_SENSOR_PROFILE "SEN0193_4CH_DECLARED")
endif()
set(TECLA_SENSOR_PROFILE "${_TECLA_DEFAULT_SENSOR_PROFILE}" CACHE STRING "Perfil compilado de hardware/capabilities")
set_property(CACHE TECLA_SENSOR_PROFILE PROPERTY STRINGS DHT22_ONLY DHT22_VEML7700 SEN0193_4CH_DECLARED)
if(TECLA_SENSOR_PROFILE STREQUAL "DHT22_ONLY")
    set(TECLA_SENSOR_PROFILE_ID 1)
elseif(TECLA_SENSOR_PROFILE STREQUAL "DHT22_VEML7700")
    set(TECLA_SENSOR_PROFILE_ID 2)
elseif(TECLA_SENSOR_PROFILE STREQUAL "SEN0193_4CH_DECLARED")
    set(TECLA_SENSOR_PROFILE_ID 4)
else()
    message(FATAL_ERROR "TECLA_SENSOR_PROFILE no reconocido: '${TECLA_SENSOR_PROFILE}'")
endif()
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/src")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(tecla_nodo)
